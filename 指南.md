# 元数据接口性能优化修改指南

## 问题分析

当前 [`/metadata/tables`](backend/api/metadata_routes.py:24) 接口存在以下性能问题：

1. **全量数据查询**：每次请求都返回所有表的完整元数据（包括所有字段信息）
2. **无缓存机制**：重复请求导致数据库资源浪费
3. **无搜索功能**：前端只能获取全量数据后自行筛选
4. **前端加载慢**：[`QueryForm.vue`](frontend/src/components/common/QueryForm.vue:351) 组件在初始化时加载所有可用表，当表数量多时性能差

## 优化方案

### 1. 后端优化

#### 1.1 添加轻量级接口参数

修改 [`get_all_table_metadata`](backend/api/metadata_routes.py:24) 函数，支持以下参数：
- `fields=false`：只返回表基本信息，不返回字段列表
- `table_name=xxx`：支持表名模糊搜索

#### 1.2 实现缓存机制

在 [`MetadataService`](backend/services/metadata_service/metadata_service.py:23) 中添加缓存：
- 使用内存缓存存储表列表数据
- 设置合理的缓存过期时间（如5分钟）
- 在数据更新时清除缓存

#### 1.3 优化数据库查询

在 [`MetadataTableRepository`](backend/repositories/metadata_repository.py:13) 中添加：
- 轻量级查询方法（不加载字段）
- 表名模糊搜索方法

### 2. 前端优化

#### 2.1 修改API调用

更新 [`metadataService.ts`](frontend/src/services/api/metadataService.ts:14) 中的 [`getTables`](frontend/src/services/api/metadataService.ts:14) 方法：
- 添加 `fields` 参数控制是否返回字段信息
- 添加 `tableName` 参数支持搜索

#### 2.2 优化组件加载

修改 [`QueryForm.vue`](frontend/src/components/common/QueryForm.vue:349) 组件：
- 初始加载时使用轻量级接口（`fields=false`）
- 实现搜索功能，减少不必要的数据传输

```python
# 在 MetadataService 类中添加新方法
def get_tables_optimized(self, is_available=None, include_fields=True, table_name_filter=None):
    """优化的表查询方法"""
    # 检查缓存
    cache_key = f"tables_{is_available}_{include_fields}_{table_name_filter}"
    cached_data = self._get_from_cache(cache_key)
    if cached_data:
        return cached_data
    
    # 根据参数选择不同的查询方法
    if table_name_filter:
        tables = self.table_repo.search_tables(table_name_filter)
    elif is_available is not None:
        if is_available == '1':
            tables = self.table_repo.get_available_tables()
        else:
            all_tables = self.table_repo.get_all()
            tables = [table for table in all_tables if table.get('is_available', 0) != 0]
    else:
        tables = self.table_repo.get_all()
    
    # 转换为字典格式
    result = []
    for table in tables:
        table_dict = {
            "id": table.id,
            "name": table.name,
            "comment": table.comment or "",
            "remark": table.remark or "",
            "is_available": int(table.is_available or 0),
            "created_at": table.created_at,
            "updated_at": table.updated_at,
        }
        
        # 根据参数决定是否包含字段
        if include_fields:
            columns_list = []
            if hasattr(table, 'columns') and table.columns:
                for column in table.columns:
                    if column is not None:
                        column_dict = {
                            "id": column.id,
                            "name": column.name,
                            "type": column.type,
                            "comment": column.comment or "",
                            "remark": column.remark or "",
                            "is_available": int(column.is_available or 0),
                            "business_type": column.business_type or "",
                            "relation_config_id": column.relation_config_id or ""
                        }
                        columns_list.append(column_dict)
            table_dict["columns"] = columns_list
        else:
            table_dict["columns"] = []
        
        result.append(table_dict)
    
    # 存入缓存
    self._set_cache(cache_key, result, timeout=300)  # 5分钟缓存
    
    return result

# 添加缓存相关方法
def _get_from_cache(self, key):
    """从缓存获取数据"""
    # 实现缓存逻辑，可以使用简单的内存字典
    return getattr(self, '_cache', {}).get(key)

def _set_cache(self, key, value, timeout=300):
    """设置缓存"""
    if not hasattr(self, '_cache'):
        self._cache = {}
    self._cache[key] = value
    # 可以添加定时清理过期缓存的逻辑

def clear_cache(self):
    """清除缓存"""
    if hasattr(self, '_cache'):
        self._cache.clear()
```

```typescript
function debounce(func: Function, delay: number) {
  let timeoutId: ReturnType<typeof setTimeout>
  return function (...args: any[]) {
    clearTimeout(timeoutId)
    timeoutId = setTimeout(() => func.apply(this, args), delay)
  }
}
```

```python
def batch_update_columns(self, table_id: int, columns_data: List[Dict[str, Any]]) -> Dict[str, Any]:
    """批量更新字段"""
    try:
        success_count = 0
        error_count = 0
        errors = []
        
        # 开始事务
        with self.db.begin():
            for column_data in columns_data:
                try:
                    # 判断是新增还是更新
                    if 'id' in column_data and column_data['id'] and not str(column_data['id']).startswith('temp_'):
                        # 更新现有字段
                        column_id = column_data['id']
                        update_data = {
                            'name': column_data.get('name'),
                            'type': column_data.get('type'),
                            'comment': column_data.get('comment'),
                            'remark': column_data.get('remark'),
                            'is_available': column_data.get('is_available', 0),
                            'business_type': column_data.get('business_type', ''),
                            'relation_config_id': column_data.get('relation_config_id')
                        }
                        # 移除None值
                        update_data = {k: v for k, v in update_data.items() if v is not None}
                        self.column_repo.update(column_id, **update_data)
                    else:
                        # 新增字段
                        self.column_repo.create(
                            table_id=table_id,
                            name=column_data.get('name'),
                            type=column_data.get('type'),
                            comment=column_data.get('comment', ''),
                            remark=column_data.get('remark', ''),
                            is_available=column_data.get('is_available', 0),
                            business_type=column_data.get('business_type', ''),
                            relation_config_id=column_data.get('relation_config_id')
                        )
                    success_count += 1
                except Exception as e:
                    error_count += 1
                    errors.append({
                        'column': column_data.get('name', 'unknown'),
                        'error': str(e)
                    })
                    logger.error(f"处理字段失败: {column_data.get('name', 'unknown')}, 错误: {e}")
        
        # 清除缓存
        self.clear_cache()
        
        return {
            'success_count': success_count,
            'error_count': error_count,
            'errors': errors
        }
    except Exception as e:
        logger.error(f"批量更新字段失败: {e}")
        raise
```

```python
def batch_update_columns_by_ids(self, table_id: int, column_ids: List[int], updates: Dict[str, Any]) -> Dict[str, Any]:
    """根据ID批量更新字段"""
    try:
        success_count = 0
        error_count = 0
        errors = []
        
        # 开始事务
        with self.db.begin():
            for column_id in column_ids:
                try:
                    # 验证字段属于该表
                    column = self.column_repo.get_by_id(column_id)
                    if not column or column.table_id != table_id:
                        error_count += 1
                        errors.append({
                            'column_id': column_id,
                            'error': '字段不存在或不属于该表'
                        })
                        continue
                    
                    # 更新字段
                    self.column_repo.update(column_id, **updates)
                    success_count += 1
                except Exception as e:
                    error_count += 1
                    errors.append({
                        'column_id': column_id,
                        'error': str(e)
                    })
                    logger.error(f"更新字段失败: ID {column_id}, 错误: {e}")
        
        # 清除缓存
        self.clear_cache()
        
        return {
            'success_count': success_count,
            'error_count': error_count,
            'errors': errors
        }
    except Exception as e:
        logger.error(f"批量更新字段失败: {e}")
        raise
```

# 多线程缓存优化方案

## 问题分析

在多线程环境下，传统的缓存方案存在以下问题：

1. **线程隔离**：每个线程有自己的缓存实例，无法共享数据
2. **缓存命中率低**：线程A写入的缓存，线程B无法命中
3. **资源浪费**：相同数据在多个线程中重复存储
4. **数据不一致**：不同线程的缓存可能存在版本差异

## TrackerCache多线程优化方案分析

### 1. 当前方案优势

从提供的 [`优化`](优化:21) 文件中的 [`TrackerCache`](优化:21) 实现来看，该方案具有以下优势：

#### 1.1 基于DiskCache的持久化缓存
```python
self.cache = DiskCache(
```

```python
import time
from functools import lru_cache

class TrackerCache:
    def __init__(self, cache_dir: str = "./task_cache", 
                 max_size: int = 100000, ttl: int = 3600, 
                 shards: int = 8, retry: int = 3,
                 memory_size: int = 1000):  # 新增内存缓存大小
        # 现有初始化代码...
        
        # 添加内存缓存层
        self._memory_cache = {}
        self._memory_cache_timestamps = {}
        self._memory_size = memory_size
        self._memory_ttl = ttl
        
    def get_task_state(self, task_id: str) -> Optional[Dict[str, Any]]:
        # 先检查内存缓存
        with self._lock:
            if task_id in self._memory_cache:
                timestamp = self._memory_cache_timestamps[task_id]
                if time.time() - timestamp < self._memory_ttl:
                    logger.debug(f"内存缓存命中: task_id={task_id}")
                    return self._memory_cache[task_id]
                else:
                    # 过期，从内存缓存中删除
                    del self._memory_cache[task_id]
                    del self._memory_cache_timestamps[task_id]
            
            # 内存缓存未命中，检查磁盘缓存
            try:
                state = self.cache.get(task_id)
                if state is not None:
                    logger.debug(f"磁盘缓存命中: task_id={task_id}")
                    # 将磁盘缓存的数据加载到内存缓存
                    self._set_memory_cache(task_id, state)
                else:
                    logger.debug(f"缓存未命中: task_id={task_id}")
                return state
            except Exception as e:
                logger.error(f"获取缓存失败: task_id={task_id}, error={e}")
                return None
    
    def _set_memory_cache(self, task_id: str, state: Dict[str, Any]):
        """设置内存缓存，LRU淘汰"""
        # 如果内存缓存已满，删除最旧的条目
        if len(self._memory_cache) >= self._memory_size:
            oldest_key = min(self._memory_cache_timestamps.keys(), 
                          key=lambda k: self._memory_cache_timestamps[k])
            del self._memory_cache[oldest_key]
            del self._memory_cache_timestamps[oldest_key]
        
        self._memory_cache[task_id] = state
        self._memory_cache_timestamps[task_id] = time.time()
```
